# GPU pipeline requirements
# torch/torchvision are not pinned — use pre-installed versions on RunPod,
# or pip will pull the latest CUDA build automatically.
# For macOS, torch/torchvision are included without CUDA builds.

torch; sys_platform == 'darwin'
torchvision; sys_platform == 'darwin'

# Core
numpy>=1.24.2,<2.0
opencv-contrib-python>=4.8.0
# insightface depends on `onnxruntime`, the CPU wheel, and pip installs it
# after onnxruntime-gpu below. Both write the same `onnxruntime/` directory,
# so the CPU build silently replaces the GPU one and CUDAExecutionProvider
# disappears. This file cannot express "resolve that dependency to the GPU
# build" - runpod/startup.sh removes the CPU wheel afterwards and verifies
# the provider is back.
insightface>=0.7.3
onnxruntime-gpu>=1.15.0; sys_platform != 'darwin'
onnxruntime==1.15.0; sys_platform == 'darwin' and platform_machine != 'arm64'
onnxruntime-silicon==1.13.1; sys_platform == 'darwin' and platform_machine == 'arm64'

# fp16 conversion (tools/convert_fp16.py). `onnx` also arrives transitively via
# insightface — INSwapper reads the emap projection out of the model file with
# it — but the conversion path needs it explicitly, and onnxconverter-common
# ships nowhere else. Both are small.
#
# These are on the *pod's* requirements because that is where the weights live:
# /workspace/models is the only place the models exist, so the conversion runs
# there. A missing dependency here is a paid hour spent on pip.
onnx>=1.14.0
onnxconverter-common>=1.13.0

# GPU memory management
tensorflow>=2.12.0

# Face restoration.
# The default backend is CodeFormer, which is ONNX and runs on the
# onnxruntime session already required above — no extra dependency, the model
# is downloaded on first use. gfpgan is only needed for the alternate
# `--enhancer-model gfpgan` backend; the pipeline runs fine without it.
gfpgan==1.3.8

# API and utilities
websockets>=12.0
python-dotenv>=1.0.0
tqdm>=4.65.0
